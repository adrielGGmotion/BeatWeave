cmake_minimum_required(VERSION 3.22)
project(beatweave_rubberband LANGUAGES C CXX)
option(BEATWEAVE_JNI "Build Java/Android bridge" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BW_SOURCES beatweave_rubberband.cpp vendor/rubberband/single/RubberBandSingle.cpp)
if(BEATWEAVE_JNI)
  list(APPEND BW_SOURCES beatweave_jni.cpp)
endif()
add_library(beatweave_rubberband SHARED ${BW_SOURCES})
target_include_directories(beatweave_rubberband PUBLIC include)
if(BEATWEAVE_JNI AND NOT ANDROID)
  find_package(JNI REQUIRED)
  target_include_directories(beatweave_rubberband PRIVATE ${JNI_INCLUDE_DIRS})
endif()
if(APPLE)
  target_link_libraries(beatweave_rubberband PRIVATE "-framework Accelerate")
endif()
# 16 KiB page-compatible Android binaries, including arm64 devices on Android 15+.
if(ANDROID)
  target_link_options(beatweave_rubberband PRIVATE "-Wl,-z,max-page-size=16384")
endif()
